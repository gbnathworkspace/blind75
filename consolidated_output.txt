﻿using Files_wpf.Models;
using Newtonsoft.Json;
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Net.Http;
using System.Net.Http.Json;
using System.Text;
using System.Text.Json.Serialization;
using System.Threading.Tasks;
using System.Windows;
using System.Xml.Linq;


namespace Files_wpf
{
    public class DriveService
    {
        private readonly HttpClient _client;

        public DriveService(string userHandle)
        {
            _client = new HttpClient();
            _client.BaseAddress = new Uri("http://localhost:5130");
            _client.DefaultRequestHeaders.Add("X-User-Handle", userHandle);
        }

        public async Task<List<DriveItemDto>> GetFolderItemsAsync(
            string folderId,
            DriveOrderBy orderBy = DriveOrderBy.ByName,
            int pageSize = 50,
            string refItemId = null,
            string refName = null,
            DateTime? refTime = null,
            long? refSize = null)
        {
            var query = $"/drive/folders/{folderId}/items?orderBy={orderBy}&pagesize={pageSize}";

            if (!string.IsNullOrEmpty(refItemId)) query += $"&refitemid={refItemId}";
            if (!string.IsNullOrEmpty(refName)) query += $"&refname={Uri.EscapeDataString(refName)}";
            if (refTime.HasValue) query += $"&reftime={refTime.Value:O}";
            if (refSize.HasValue) query += $"&refsize={refSize.Value}";

            return await _client.GetFromJsonAsync<List<DriveItemDto>>(query);
        }

        public async Task<List<DriveItemDto>> GetTrashItemsAsync()
        {
            return await _client.GetFromJsonAsync<
                List<DriveItemDto>>
            (
                "/trash"
            );
        }

        public async Task CreateFolderAsync(string parentId, string folderName)
        {
            var request = new
            {
                parentId = parentId,
                folderName = folderName
            };

            await _client.PostAsJsonAsync(
                "/drive/folders",
                request);
        }

        public async Task<Tuple<string, string>> CreateFileAsync(string parentId, string fileName, string previewData, string mimeType, long fileSize, S3Object s3Objects)
        {
            var request = new
            {
                parentId = parentId,
                fileName = fileName,
                previewData = previewData,
                mimeType = mimeType,
                fileSize = fileSize,
                s3Objects = s3Objects
            };

            var response = await _client.PostAsJsonAsync("/drive/files", request);
            response.EnsureSuccessStatusCode();

            var json = await response.Content.ReadAsStringAsync();
            var result = JsonConvert.DeserializeObject<CreateFileResponse>(json);

            return Tuple.Create(result.itemid, result.versionid);
        }

        private class CreateFileResponse
        {
            public string itemid { get; set; }
            public string versionid { get; set; }
        }

        public async Task DeleteFileAsync(string fileId)
        {
            await _client.DeleteAsync(
                $"/drive/files/{fileId}");
        }

        public async Task UploadFileAsync(string filePath, string parentId)
        {
            using (var stream = File.OpenRead(filePath))
            using (var content = new MultipartFormDataContent())
            {
                content.Add(new StreamContent(stream), "file", Path.GetFileName(filePath));
                await _client.PostAsync($"/drive/files/upload?parentId={parentId}", content);
            }
        }


        public async Task<List<S3Object>> GetS3ObjectsAsync(List<DriveItemsInput> input)
        {
            var json = JsonConvert.SerializeObject(input);
            var content = new StringContent(json, Encoding.UTF8, "application/json");

            var response = await _client.PostAsync("/preview/s3", content);
            response.EnsureSuccessStatusCode();

            var resultJson = await response.Content.ReadAsStringAsync();
            return JsonConvert.DeserializeObject<List<S3Object>>(resultJson);
        }

        public async Task<string> GetPreviewDataAsync(string itemId)
        {
            var response = await _client.GetAsync($"/preview/{itemId}");
            response.EnsureSuccessStatusCode();

            var json = await response.Content.ReadAsStringAsync();

            var result = JsonConvert.DeserializeObject<PreviewResponse>(json);
            return result?.previewdata;
        }

        private class PreviewResponse
        {
            public string previewdata { get; set; }
        }
    }
}

﻿using ClaritiFiles.Helpers;
using ClaritiFiles.Views;
using Files_wpf.Models;
using System;
using System.Collections.Generic;
using System.Collections.ObjectModel;
using System.Diagnostics;
using System.IO;
using System.Net.Http;
using System.Threading.Tasks;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Input;
using System.Windows.Media;
using static System.Net.WebRequestMethods;

namespace Files_wpf
{
    public partial class MainWindow : Window
    {
        private ObservableCollection<DriveItemDto> files = new ObservableCollection<DriveItemDto>();

        private ObservableCollection<TrashItem> trashItems =
            new ObservableCollection<TrashItem>();

        private ObservableCollection<ShareItem> shareItems =
            new ObservableCollection<ShareItem>();

        static string userhandle = "d7d648ff-b2e6-4eb0-9fc5-4cfaabe73a43";

        DriveService _driveService = new DriveService(userhandle);

        private string _currentFolderId = userhandle;

        public MainWindow()
        {
            InitializeComponent();

            //SeedData();

            FileListView.ItemsSource = files;
            TrashListView.ItemsSource = trashItems;
            SharedItemView.ItemsSource = shareItems;

            Loaded += async (s, e) => await LoadFilesAsync();
        }

        private async Task LoadFilesAsync()
        {
            try
            {
                var items = await _driveService.GetFolderItemsAsync(_currentFolderId);

                files.Clear();
                foreach (var item in items)
                {
                    files.Add(new DriveItemDto
                    {
                        ItemId = item.ItemId,
                        Name = item.Name,
                        ModifiedTime = item.ModifiedTime,
                        IsFolder = item.IsFolder,
                        Size = item.Size
                    });
                }
            }

            catch (Exception ex)
            {
                MessageBox.Show($"Failed to load files: {ex.Message}", "Error",
                    MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        private void AddButton_Click(object sender, RoutedEventArgs e)
        {
            ContextMenu menu = new ContextMenu();

            MenuItem addFile = new MenuItem();
            addFile.Header = "Add File";
            addFile.Click += AddFile_Click;

            MenuItem addFolder = new MenuItem();
            addFolder.Header = "Add Folder";
            addFolder.Click += AddFolder_Click;

            menu.Items.Add(addFile);
            menu.Items.Add(addFolder);

            menu.PlacementTarget = AddButton;
            menu.IsOpen = true;
        }

        private async void AddFile_Click(object sender, RoutedEventArgs e)
        {
            var dialog = new Microsoft.Win32.OpenFileDialog
            {
                Title = "Select a file",
                Multiselect = false
            };

            if (dialog.ShowDialog() == true)
            {
                var info = new FileInfo(dialog.FileName);

                try
                {
                    // Use the currently tracked folder id when uploading.
                    await _driveService.UploadFileAsync(dialog.FileName, parentId: _currentFolderId);

                    // Refresh the view from the server so uploaded file (and server-assigned ids) are shown.
                    await LoadFilesAsync();
                }
                catch (Exception ex)
                {
                    MessageBox.Show($"Upload failed: {ex.Message}", "Error",
                        MessageBoxButton.OK, MessageBoxImage.Error);
                }
            }
        }

        private string FormatFileSize(long bytes)
        {
            if (bytes < 1024) return $"{bytes} B";
            if (bytes < 1024 * 1024) return $"{bytes / 1024.0:F1} KB";
            if (bytes < 1024 * 1024 * 1024) return $"{bytes / (1024.0 * 1024):F1} MB";
            return $"{bytes / (1024.0 * 1024 * 1024):F1} GB";
        }

        private async void AddFolder_Click(object sender, RoutedEventArgs e)
        {
            try
            {
                // Create a new folder on the server under the currently tracked folder.
                await _driveService.CreateFolderAsync(_currentFolderId, "New Folder");

                // Refresh the view to show the newly created folder.
                await LoadFilesAsync();
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Create folder failed: {ex.Message}", "Error",
                    MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        private void DeleteMenuItem_Click(object sender, RoutedEventArgs e)
        {
            DriveItemDto selected =
                FileListView.SelectedItem as DriveItemDto;

            if (selected == null)
                return;

            files.Remove(selected);

            trashItems.Add(new TrashItem
            {
                Name = selected.Name,
                ModifiedTime = selected.ModifiedTime,
                Size = selected.Size
            });
        }

        private void ShareMenuItem_Click(object sender, RoutedEventArgs e)
        {
            DriveItemDto selected =
                FileListView.SelectedItem as DriveItemDto;

            if (selected == null)
                return;

            files.Remove(selected);

            shareItems.Add(new ShareItem
            {
                Name = selected.Name,
                ModifiedTime = selected.ModifiedTime,
                Size = selected.Size
            });
        }

        private void RestoreMenuItem_Click(object sender, RoutedEventArgs e)
        {
            TrashItem selected =
                TrashListView.SelectedItem as TrashItem;

            if (selected == null)
                return;

            trashItems.Remove(selected);

            files.Add(new DriveItemDto
            {
                Name = selected.Name,
                ModifiedTime = selected.ModifiedTime,
                Size = Convert.ToInt64(selected.Size)
            });
        }

        private void DoubleClickFile(object sender, MouseButtonEventArgs e)
        {
            if (e.ChangedButton != MouseButton.Left) return;

            var source = e.OriginalSource as DependencyObject;
            while (source != null && !(source is ListViewItem) && !(source is GridViewColumnHeader))
                source = VisualTreeHelper.GetParent(source);

            if (source is GridViewColumnHeader) return;
            var lvi = source as ListViewItem;
            if (lvi == null) return;

            Debug.WriteLine($"[DoubleClick] DataContext type: {lvi.DataContext?.GetType().Name}");

            DriveItemDto selected = lvi.DataContext as DriveItemDto;
            if (selected == null) return;

            Debug.WriteLine($"[DoubleClick] {selected.Name}");
            try
            {
                if (selected.IsFolder)
                    OpenFolder(selected);
                else
                    OpenFile(selected);
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Could not open '{selected.Name}'.\n\n{ex.Message}",
                    "Error", MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        private async void OpenFile(DriveItemDto file)
        {
            try
            {
                // Call both in parallel
                var previewTask = _driveService.GetPreviewDataAsync(file.ItemId);
                var s3Task = _driveService.GetS3ObjectsAsync(new List<DriveItemsInput>
                {
                    new DriveItemsInput { ItemId = file.ItemId, OwnerHandle = userhandle }
                });

                await Task.WhenAll(previewTask, s3Task);

                string previewData = previewTask.Result;
                var s3Objects = s3Task.Result;

                if (s3Objects == null || s3Objects.Count == 0)
                {
                    MessageBox.Show($"File storage location not found for '{file.Name}'.",
                        "Preview Error", MessageBoxButton.OK, MessageBoxImage.Warning);
                    return;
                }

                PreviewWindowLauncher.Open(
                    itemId: file.ItemId,
                    fileName: file.Name,
                    previewData: previewData,
                    s3Objects: s3Objects,
                    owner: this);
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Could not open '{file.Name}'.\n\n{ex.Message}",
                    "Preview Error", MessageBoxButton.OK, MessageBoxImage.Warning);
                System.Diagnostics.Debug.WriteLine($"[OpenFile] {file.Name} | {ex}");
            }
        }

        private async Task OpenFolder(DriveItemDto folder)
        {
            var items = await _driveService.GetFolderItemsAsync(folder.ItemId);
            FileListView.ItemsSource = items;
        }
    }
}

using Amazon.S3;
using Amazon.S3.Model;
using Files_wpf.Models;
using System;
using System.Diagnostics;
using System.IO;
using System.Net.Http;
using System.Threading.Tasks;
using System.Windows;
using System.Windows.Media.Animation;
using System.Windows.Media.Imaging;

namespace ClaritiFiles.Views
{
    /// <summary>
    /// Online file preview window.
    /// Routing logic (think: a switch on file extension):
    ///   .pdf              → WebBrowser (streams PDF from API)
    ///   .jpg/.png/.gif    → WPF Image  (streams bitmap from API)
    ///   .txt/.cs/.json…   → WebBrowser (renders raw text / code)
    ///   .docx/.xlsx/.pptx → Big icon  (no server-side conversion)
    ///   anything else     → Big icon
    /// </summary>
    public partial class PreviewWindow : Window
    {
        // ── Config ────────────────────────────────────────────────────────────
        private const string ApiBase = "http://localhost:5130";

        // ── State ─────────────────────────────────────────────────────────────
        private readonly string _itemId;
        private readonly string _fileName;

        // Lazy HttpClient shared per window (not app-wide — window is short-lived)
        private static readonly HttpClient _http = new HttpClient { Timeout = TimeSpan.FromSeconds(30) };

        private readonly Files_wpf.Models.S3Object _s3Object;
        private readonly string _previewData;
        private static readonly AmazonS3Client _s3Client = new AmazonS3Client();

        // ─────────────────────────────────────────────────────────────────────

        public string FileName => _fileName;

        public PreviewWindow(string itemId, string fileName, string previewData, Files_wpf.Models.S3Object s3Object)
        {
            _itemId = itemId ?? throw new ArgumentNullException(nameof(itemId));
            _fileName = fileName ?? "Preview";
            _previewData = previewData;
            _s3Object = s3Object ?? throw new ArgumentNullException(nameof(s3Object));

            InitializeComponent();
            DataContext = this;
            SuppressBrowserErrors();
        }

        // ── Lifecycle ─────────────────────────────────────────────────────────

        protected override void OnContentRendered(EventArgs e)
        {
            base.OnContentRendered(e);
            _ = LoadPreviewAsync();   // fire-and-forget (async void pattern avoided)
        }

        // ── Core routing ──────────────────────────────────────────────────────

        private async Task LoadPreviewAsync()
        {
            var ext = Path.GetExtension(_fileName).ToLowerInvariant();

            // Update toolbar icon immediately (no network round-trip needed)
            TbFileIcon.Text = GetToolbarIcon(ext);
            SetStatus("Loading…");
            ShowPanel(PanelLoading);

            try
            {
                switch (GetFileCategory(ext))
                {
                    case FileCategory.Pdf:
                        await LoadBrowserAsync(ext);
                        break;

                    case FileCategory.Image:
                        await LoadImageAsync();
                        break;

                    case FileCategory.Text:
                        await LoadBrowserAsync(ext);
                        break;

                    case FileCategory.Office:
                    case FileCategory.Unknown:
                    default:
                        ShowOfficeIcon(ext);
                        break;
                }
            }
            catch (Exception ex)
            {
                ShowError($"Could not load preview.\n{ex.Message}");
            }
        }

        // ── Loaders ───────────────────────────────────────────────────────────

        /// <summary>
        /// PDF and text files: hand the preview URL directly to WebBrowser.
        /// The browser will stream and render it natively.
        /// </summary>
        private Task LoadBrowserAsync(string ext)
        {
            var url = $"{ApiBase}/preview/{Uri.EscapeDataString(_itemId)}";
            BrowserViewer.Navigate(url);
            BrowserViewer.LoadCompleted += (_, __) =>
            {
                SetStatus($"Loaded  ·  {_fileName}");
            };

            ShowPanel(BrowserViewer);
            SetStatus($"Streaming {ext.TrimStart('.')} preview…");
            return Task.CompletedTask;
        }

        /// <summary>
        /// Images: download bytes → BitmapImage (keeps memory clean, no temp files).
        /// </summary>
        private async Task LoadImageAsync()
        {
            var url   = $"{ApiBase}/preview/{Uri.EscapeDataString(_itemId)}";
            var bytes = await _http.GetByteArrayAsync(url);

            var bmp = new BitmapImage();
            using (var ms = new MemoryStream(bytes))
            {
                bmp.BeginInit();
                bmp.StreamSource    = ms;
                bmp.CacheOption     = BitmapCacheOption.OnLoad;
                bmp.EndInit();
                bmp.Freeze();   // cross-thread safe
            }

            ImageViewer.Source = bmp;

            // File size in status bar
            TbFileSize.Text = FormatBytes(bytes.Length);
            SetStatus($"Image loaded  ·  {bmp.PixelWidth} × {bmp.PixelHeight} px");
            ShowPanel(ImageScroll);
        }

        /// <summary>
        /// Office / unsupported: show the big animated icon.
        /// </summary>
        private void ShowOfficeIcon(string ext)
        {
            TbBigIcon.Text   = GetBigIcon(ext);
            TbIconLabel.Text = ext.TrimStart('.').ToUpperInvariant() + " Document";
            ShowPanel(PanelIcon);

            // Gentle pulse animation — feels alive without being distracting
            var pulse = new DoubleAnimation(1.0, 1.08, TimeSpan.FromSeconds(0.9))
            {
                AutoReverse  = true,
                RepeatBehavior = new RepeatBehavior(2)
            };
            IconScale.BeginAnimation(System.Windows.Media.ScaleTransform.ScaleXProperty, pulse);
            IconScale.BeginAnimation(System.Windows.Media.ScaleTransform.ScaleYProperty, pulse);

            SetStatus($"No preview available for {ext} files");
        }

        // ── UI helpers ────────────────────────────────────────────────────────

        private void ShowPanel(UIElement target)
        {
            // Hide all content panels, then show the requested one
            UIElement[] panels =
            {
                PanelLoading, PanelError, BrowserViewer,
                ImageScroll, PanelIcon
            };

            foreach (var p in panels)
                p.Visibility = ReferenceEquals(p, target) ? Visibility.Visible : Visibility.Collapsed;
        }

        private void ShowError(string message)
        {
            TbErrorMessage.Text = message;
            ShowPanel(PanelError);
            SetStatus("Error");
        }

        private void SetStatus(string text) => TbStatus.Text = text;

        // ── Toolbar handlers ──────────────────────────────────────────────────

        private void BtnRefresh_Click(object sender, RoutedEventArgs e)
            => _ = LoadPreviewAsync();

        private void BtnOpenExternal_Click(object sender, RoutedEventArgs e)
        {
            // Opens the preview URL in the system's default browser / viewer
            var url = $"{ApiBase}/preview/{Uri.EscapeDataString(_itemId)}";
            Process.Start(url);   // .NET Framework: Process.Start(string) opens in default browser
        }

        private void BtnClose_Click(object sender, RoutedEventArgs e) => Close();

        // ── Classification ────────────────────────────────────────────────────

        private enum FileCategory { Pdf, Image, Text, Office, Unknown }

        private static FileCategory GetFileCategory(string ext)
        {
            switch (ext)
            {
                case ".pdf":
                    return FileCategory.Pdf;

                case ".jpg":
                case ".jpeg":
                case ".png":
                case ".gif":
                case ".bmp":
                case ".webp":
                    return FileCategory.Image;

                case ".txt":
                case ".md":
                case ".cs":
                case ".ts":
                case ".js":
                case ".json":
                case ".xml":
                case ".yaml":
                case ".yml":
                case ".log":
                case ".csv":
                    return FileCategory.Text;

                case ".docx":
                case ".doc":
                case ".xlsx":
                case ".xls":
                case ".pptx":
                case ".ppt":
                    return FileCategory.Office;

                default:
                    return FileCategory.Unknown;
            }
        }

        // Small toolbar icon (emoji works fine in WPF TextBlock)
        private static string GetToolbarIcon(string ext)
        {
            switch (ext)
            {
                case ".pdf":                        return "📕";
                case ".jpg": case ".jpeg":
                case ".png": case ".gif":           return "🖼";
                case ".docx": case ".doc":          return "📘";
                case ".xlsx": case ".xls":          return "📗";
                case ".pptx": case ".ppt":          return "📙";
                case ".txt": case ".md":            return "📝";
                case ".json": case ".xml":
                case ".yaml": case ".yml":          return "🔧";
                case ".cs": case ".ts": case ".js": return "💻";
                default:                            return "📄";
            }
        }

        // Big icon shown in the icon panel (80px TextBlock)
        private static string GetBigIcon(string ext)
        {
            switch (ext)
            {
                case ".docx": case ".doc":  return "📘";
                case ".xlsx": case ".xls":  return "📗";
                case ".pptx": case ".ppt":  return "📙";
                case ".zip":  case ".rar":  return "🗜";
                default:                    return "📄";
            }
        }

        // ── Utilities ─────────────────────────────────────────────────────────

        private static string FormatBytes(long bytes)
        {
            if (bytes < 1024)       return $"{bytes} B";
            if (bytes < 1048576)    return $"{bytes / 1024.0:F1} KB";
            return                         $"{bytes / 1048576.0:F1} MB";
        }

        /// <summary>
        /// Suppress the "Script error" dialog that WPF's IE-based WebBrowser raises
        /// when loading modern pages or PDF viewer scripts.
        /// </summary>
        private void SuppressBrowserErrors()
        {
            BrowserViewer.Navigating += (_, e) =>
            {
                // No-op — hook kept as extension point for auth headers
            };

            // Silence IE scripting errors via COM reflection
            BrowserViewer.LoadCompleted += (s, _) =>
            {
                try
                {
                    dynamic doc = ((System.Windows.Controls.WebBrowser)s).Document;
                    doc?.parentWindow?.execScript("window.onerror = function(){return true;}", "javascript");
                }
                catch { /* best-effort */ }
            };
        }

    /// <summary>

        /// Opens the preview window for a given drive item.
        /// </summary>

        public static void Open(string itemId, string fileName, string previewData, List<S3Object> s3Objects, Window owner)
        {
            var s3Object = s3Objects.FirstOrDefault(o => /* match if you have multiple per item, else just take first */ true);
            if (s3Object == null)
            {
                MessageBox.Show($"No storage object found for '{fileName}'.", "Preview Error",
                    MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            var win = new PreviewWindow(itemId, fileName, previewData, s3Object);
            if (owner != null)
                win.Owner = owner;
